\begin{abstract}
Pre-trained autoregressive Transformers exhibit an effective ``Event Horizon'' during long-context multi-hop reasoning: even when Rotary Position Embeddings (RoPE) are pre-trained for $128\text{K}+$ token windows and achieve $>99.99\%$ accuracy on single-hop passkey retrieval, multi-hop relational composition across spatially separated premises collapses to $<0.11\%$ accuracy in the presence of late-context adversarial decoys. We present a rigorous statistical-mechanical theory and parameter-free runtime inference architecture---\textbf{Richardson Attention (Superconducting Spectral Attention)}---that resolves this collapse without weight modification or retraining. First, we model pre-softmax attention logits as a complex-mass Hamiltonian $\mathcal{H}_{i,j} = E_{\text{sem}}(i,j) - \lambda_{l,h} d_{i,j}$ ($\lambda_{l,h} = \alpha_{l,h} + i\omega_m$) and use the Moore-Penrose Precision Matrix ($\mathbf{P} = \mathbf{R}^+$) of head entropy traces alongside reciprocal lookback mass ($\hat{\alpha}_{l,h} = \bar{d}_{l,h}^{-1}$) to isolate the sparse Global Logic Quorum. Second, we prove the \emph{Write/Read Thermodynamic Asymmetry Theorem}, showing that KV-cache construction during document prefill requires local gravitational confinement ($d \le W_{\text{local}}$) to preserve intra-sentence entity binding, whereas active query readout ($W_q = 64$) requires superconducting decay cancellation. Third, we introduce \emph{Dual-State Symmetric Boltzmann Extrapolation} coupled with an \emph{Iso-Potential Cubic Plateau} and a \emph{Depth-Progressive Renormalization Group (RG) Schedule} across Global Attention layers. Evaluated on unmodified \texttt{google/gemma-4-E2B-it} and \texttt{google/gemma-4-E4B-it} models up to $11,975$ tokens, our architecture preserves baseline syntactic perplexity ($\Delta\text{PPL} = +0.01$), maintains $38:1$ RoPE phase selectivity between adjacent axioms, suppresses adversarial decoy leakage by $485\times\text{--}1,427\times$, and flips 3-hop pointer-chasing accuracy from $0.038\%$ (Baseline Fail) to $98.59\%$ (Rank \#1 Win, a $2,594\times$ probability gain).
\end{abstract}

\section{Complex Spectral Mass and the Attention Partition Function}
% ============================================================================

\subsection{The Complex Positional Propagator}
Let $\mathcal{M}$ be an $L$-layer autoregressive Transformer with $H$ query heads per layer and head dimension $d_k$. For a sequence of length $T$, let $q_i, k_j \in \mathbb{R}^{d_k}$ denote the pre-rotational query and key projections at causal token positions $i \ge j \ge 0$, with causal lookback distance $d_{i,j} = \max(0, i - j)$.

In architectures employing Rotary Position Embeddings (RoPE or Proportional RoPE) alongside explicit or implicit distance attenuation, the effective attention kernel between query $i$ and key $j$ in layer $l$, head $h$ can be written in complex spectral form:
\begin{equation}
    \mathcal{K}^{(l,h)}(i, j) = \sum_{m=1}^{d_k / 2} \tilde{q}_{i,m}^* \tilde{k}_{j,m} \exp\!\big(-\lambda_{l,h,m}\, d_{i,j}\big), \qquad \lambda_{l,h,m} = \alpha_{l,h} + i\omega_m,
    \label{eq:complex_mass}
\end{equation}
where $\tilde{q}_{i,m}, \tilde{k}_{j,m} \in \mathbb{C}$ are the complexified 2D rotary pairs, $\omega_m = \theta_{\text{base}}^{-2(m-1)/d_k}$ is the rotational phase frequency, and $\alpha_{l,h} \ge 0$ is the effective real spectral mass (inverse correlation length $\xi_{l,h}^{-1}$) induced during pre-training.

Taking the log-modulus of the positional envelope yields the pre-softmax statistical-mechanical Hamiltonian (logit matrix) $L^{(l,h)} \in \mathbb{R}^{T \times T}$:
\begin{equation}
    L^{(l,h)}_{i,j} = E_{\text{sem}}^{(l,h)}(i, j;\, \omega) - \alpha_{l,h}\, d_{i,j} + M_{\text{causal}}(i, j),
    \label{eq:hamiltonian}
\end{equation}
where $E_{\text{sem}}^{(l,h)}(i, j;\, \omega) = \frac{1}{\sqrt{d_k}} q_i^\top R_{\Theta, i-j} k_j$ is the phase-modulated semantic resonance and $M_{\text{causal}}(i,j) \in \{0, -\infty\}$ enforces causality (and local window truncation $d_{i,j} \le W_{\text{SWA}}$ on Sliding-Window layers).

\subsection{Boltzmann Partition Dilution and the Multi-Hop Event Horizon}
The post-softmax attention distribution $A^{(l,h)}_{i,j}$ is the Boltzmann Gibbs measure over the causal past of token $i$:
\begin{equation}
    A^{(l,h)}_{i,j} = \frac{\exp\!\big(L^{(l,h)}_{i,j}\big)}{Z^{(l,h)}_i}, \qquad Z^{(l,h)}_i = \sum_{r=0}^{i} \exp\!\Big(E_{\text{sem}}^{(l,h)}(i, r;\, \omega) - \alpha_{l,h}\, d_{i,r}\Big).
    \label{eq:boltzmann}
\end{equation}

\begin{proposition}[The Multi-Hop Event Horizon]
Suppose a reasoning task requires composing an $M$-hop relational chain across spatially separated registries $\mathcal{Z}_1, \mathcal{Z}_2, \dots, \mathcal{Z}_M$ at lookback distances $d_1 < d_2 < \dots < d_M$ in the presence of $N_{\text{decoy}}$ late-context adversarial decoys at distances $d_{\text{decoy}} \ll d_1$ with comparable semantic resonance $E_{\text{sem}}(i, r_{\text{decoy}}) \approx E_{\text{sem}}(i, j_m)$. Then the signal-to-decoy attention ratio for hop $m$ decays as:
\begin{equation}
    \frac{A^{(l,h)}(i \to \mathcal{Z}_m)}{A^{(l,h)}(i \to \mathcal{Z}_{\text{decoy}})} \propto \frac{1}{N_{\text{decoy}}} \exp\!\Big(-\alpha_{l,h}\big(d_m - \bar{d}_{\text{decoy}}\big)\Big).
\end{equation}
Across an $M$-hop vertical relay over layers $l_1 < \dots < l_M$, the joint probability of resolving the true terminal target without residual-stream overwrite by the decoy attractor scales as $\prod_{m=1}^M \big(1 + N_{\text{decoy}} e^{\alpha_{l_m, h}(d_m - \bar{d}_{\text{decoy}})}\big)^{-1}$, vanishing exponentially once $d_m - \bar{d}_{\text{decoy}} > \alpha_{l_m,h}^{-1} \ln N_{\text{decoy}}$.
\end{proposition}

\subsection{Boundary Pinning: Sink Protection and Local Syntactic Guardband}
Two boundary conditions govern safe runtime perturbation of Eq.~\eqref{eq:hamiltonian}:
\begin{enumerate}
    \item \textbf{Token-0 Attention Sink Protection ($j = 0$):} Autoregressive Transformers allocate excess probability mass to the initial \texttt{<bos>} token ($j = 0$) as an attention sink. Because $j = 0$ maximizes $d_{i,0} = i$, any naive distance-proportional boost amplifies the uninformative sink. We therefore pin the sink distance to zero: $d_{i,0} \equiv 0$ for all $i > 32$.
    \item \textbf{Local Syntactic Guardband ($W_{\text{guard}} = 32$):} Local syntactic agreement and subword composition occur within $d_{i,j} \le 32$. We define the long-range causal distance operator:
    \begin{equation}
        d_{\text{long}}(i, j) = \begin{cases}
            0, & \text{if } j = 0 \text{ and } i > 32, \\
            \max(0,\, i - j - W_{\text{guard}}), & \text{otherwise.}
        \end{cases}
        \label{eq:d_long}
    \end{equation}
\end{enumerate}

\subsection{Effective Spectral Mass and Precision-Matrix Agency}
To prevent syntactic aphasia (Claim 3), spectral extrapolation must never be applied blindly to high-mass syntactic heads. We isolate the network's long-range causal driver heads in a single zero-shot calibration pass of length $H_{\text{cal}}$ using two dual observables:

\begin{definition}[Empirical Spectral Mass $\hat{\alpha}_{l,h}$]
For each head $k = (l, h)$ with post-softmax attention matrix $A^{(l,h)} \in \mathbb{R}^{H_{\text{cal}} \times H_{\text{cal}}}$, the mean causal lookback distance $\bar{d}_{l,h}$ and Effective Spectral Mass $\hat{\alpha}_{l,h}$ are:
\begin{equation}
    \bar{d}_{l,h} = \frac{1}{H_{\text{cal}}} \sum_{i=0}^{H_{\text{cal}}-1} \sum_{j=0}^{i} A^{(l,h)}_{i,j}\, |i - j|, \qquad \hat{\alpha}_{l,h} = \frac{1}{\bar{d}_{l,h} + \epsilon}, \qquad \kappa_{l,h} = \operatorname{clip}\!\big(\hat{\alpha}_{l,h} H_{\text{cal}},\, \kappa_{\min},\, \kappa_{\max}\big),
    \label{eq:eff_mass}
\end{equation}
where $\kappa_{l,h} \in [1.8, 2.6]$ is the dimensionless calibrated coupling constant.
\end{definition}

\begin{definition}[Precision-Matrix Causal Agency via Leinster Duality]
For each head $k \in \{1, \dots, L H\}$, let $H_k \in \mathbb{R}^{H_{\text{cal}}}$ denote its per-token Shannon entropy trajectory:
\begin{equation}
    H_k(i) = -\sum_{j=0}^{i} A^{(k)}_{i,j} \ln\!\big(A^{(k)}_{i,j} + \epsilon\big).
\end{equation}
Stacking the standardized trajectories into $\mathbf{X} \in \mathbb{R}^{H_{\text{cal}} \times (LH)}$, let $\mathbf{R} = \operatorname{corrcoef}(\mathbf{X}) \in \mathbb{R}^{(LH) \times (LH)}$ be the inter-head functional connectivity matrix. By Leinster Duality (inverting geometric similarity into logical precision), the Precision Matrix is the Moore-Penrose pseudoinverse $\mathbf{P} = \mathbf{R}^+$, and the \textbf{Causal Agency} of head $k$ is its diagonal entry:
\begin{equation}
    \operatorname{Agency}(k) = P_{k,k} = \frac{1}{1 - R_k^2},
    \label{eq:agency}
\end{equation}
where $R_k^2$ is the coefficient of multiple determination of head $k$'s entropy trace regressed against all other $LH - 1$ heads. High $P_{k,k}$ combined with low $\hat{\alpha}_{l,h}$ identifies heads that act as global information hubs across the entire context window.
\end{definition}

\section{The Write/Read Thermodynamic Asymmetry Theorem}
% ============================================================================

A foundational discovery of our Phase 4 and Phase 5 ablation experiments is why applying backward distance extrapolation to \emph{all} causal rows $i \in [0, T)$ during prefill degrades intermediate multi-hop entity binding, whereas restricting extrapolation to the active query slice $i \in [T - W_q, T)$ restores $>99.5\%$ 2-hop accuracy and unlocks $>95\%$ 3-hop accuracy.

\begin{theorem}[Write/Read Phase Asymmetry in Autoregressive Memory]
\label{thm:write_read}
Consider a relational entry in an intermediate registry $\mathcal{Z}_m$ spanning token positions $[s_m, e_m]$ (with intra-entry span $\Delta_{\text{bind}} = e_m - s_m \le W_{\text{guard}}$), such as $\text{\texttt{SECTOR-KRONOS}} \mapsto \text{\texttt{VAULT-LAZARUS}}$ at tokens $[1728, 1749]$.
\begin{enumerate}
    \item \textbf{The KV-Write Phase (Prefill Encoding, $i \in [s_m, e_m]$):} For the contextualized hidden state $x_{e_m}^{(l)}$ (and its resulting Key/Value projections $k_{e_m}^{(l+1)}, v_{e_m}^{(l+1)}$) to bind the local subject token at $s_m$ into the object token at $e_m$, the attention mass at row $i = e_m$ must concentrate within the local clause $[s_m, e_m]$ ($d_{e_m, s_m} \le W_{\text{guard}}$). If a positive long-range potential $+\Delta L > 0$ is injected at prefill row $i = e_m$ for all $j < e_m - W_{\text{guard}}$, the partition function $Z_{e_m}$ is inflated by $\Delta Z_{e_m} \approx \sum_{j=1}^{s_m - 1} e^{L_{e_m, j} + \Delta L}$, attenuating the intra-clause binding probability $A_{e_m, s_m}$ by the factor:
    \begin{equation}
        \frac{A^{\text{tilted}}_{e_m, s_m}}{A^{\text{base}}_{e_m, s_m}} = \frac{1}{1 + \big(1 - P_{\text{local}}\big)\big(e^{\overline{\Delta L}} - 1\big)} \sim \mathcal{O}\!\big(e^{-\beta \kappa}\big).
    \end{equation}
    Consequently, tilting document rows during prefill corrupts the relational Value vectors written into the KV cache.
    \item \textbf{The KV-Read Phase (Query Readout, $i \in [T - W_q, T)$):} Once all document registries $\mathcal{Z}_1, \dots, \mathcal{Z}_M$ have been encoded into the KV cache under unmodified local gravity ($\Delta L_{i, \cdot} \equiv 0$ for $i < T - W_q$), applying spectral extrapolation exclusively on the query readout slice $i \in [T - W_q, T)$ amplifies long-range retrieval across the cleanly bound KV pairs without retroactively distorting their stored representations.
\end{enumerate}
\end{theorem}

\section{Dual-State Symmetric Extrapolation and the Iso-Potential Cubic Plateau}
% ============================================================================

\subsection{Overcoming the \texorpdfstring{$2S_1$}{2S1} Algebraic Ceiling via Dual-State Boltzmann Extrapolation}
In classical first-order post-softmax Richardson extrapolation, one combines the baseline distribution $S_1 = \operatorname{softmax}(L)$ with a double-decayed distribution $S_2 = \operatorname{softmax}(L - \Delta L)$ via $S_{\text{Rich}} = 2 S_1 - S_2$. While this cancels first-order $\mathcal{O}(\alpha)$ partition decay, it is bounded above by $2 S_1$: as $\Delta L \to \infty$, $S_2 \to 0$ on deep tokens and the amplification saturates at $2\times$ (observed empirically in Phases 1 and 2, where post-softmax extrapolation plateaued at $+18.5\%\text{--}22.1\%$ relative attention gain).

To eliminate the $2 S_1$ ceiling while retaining active decoy subtraction, we evaluate the Boltzmann distribution at two symmetric perturbations $\pm \Delta L^{(l,h)}_{i,j}$ around the base Hamiltonian:
\begin{align}
    S_{\text{free}}^{(l,h)}(i, \cdot) &= \operatorname{softmax}\!\Big(L^{(l,h)}_{i, \cdot} + \beta \kappa_{l,h}\, \Phi_l(i, \cdot)\Big), \label{eq:s_free} \\
    S_{\text{heavy}}^{(l,h)}(i, \cdot) &= \operatorname{softmax}\!\Big(L^{(l,h)}_{i, \cdot} - \beta \kappa_{l,h}\, \Phi_l(i, \cdot)\Big), \label{eq:s_heavy} \\
    S_{\text{Super}}^{(l,h)}(i, \cdot) &= \mathcal{N}\!\left(\max\!\Big(0, \, (1 + \eta)\, S_{\text{free}}^{(l,h)}(i, \cdot) - \eta\, S_{\text{heavy}}^{(l,h)}(i, \cdot)\Big)\right), \label{eq:s_super}
\end{align}
where $\mathcal{N}(v) = v / (\sum_j v_j + \epsilon)$ is row-normalization and $\eta \in [0, 1]$ is the decoy-subtraction coefficient.

\begin{remark}[Matched-Filter Decoy Subtraction ($-\eta S_{\text{heavy}}$)]
When gravity is doubled in Eq.~\eqref{eq:s_heavy}, probability mass drains out of the deep registries $\mathcal{Z}_1, \dots, \mathcal{Z}_M$ and concentrates onto the late-context adversarial decoys $\mathcal{Z}_{\text{decoy}}$. Subtracting $-\eta S_{\text{heavy}}$ in Eq.~\eqref{eq:s_super} therefore acts as a self-calibrating causal notch filter that carves out recent confounding associations while $S_{\text{free}}$ exponentially lifts the deep registries.
\end{remark}

\subsection{RoPE Phase Invariance (Zero Spatial Smearing)}
Because $\Delta L^{(l,h)}_{i,j} = \beta \kappa_{l,h} \Phi_l(i,j)$ is purely real and varies smoothly with distance $d_{i,j}$, adding $\Delta L^{(l,h)}_{i,j}$ inside the Boltzmann exponent scales any two adjacent tokens $j_1, j_2$ at comparable lookback distance ($d_{i,j_1} \approx d_{i,j_2}$) by the same common factor $\exp(\Delta L)$, preserving their relative odds ratio:
\begin{equation}
    \frac{S_{\text{free}}(i, j_1)}{S_{\text{free}}(i, j_2)} = \exp\!\Big(E_{\text{sem}}(i, j_1;\, \omega) - E_{\text{sem}}(i, j_2;\, \omega)\Big) = \frac{A_{\text{base}}(i, j_1)}{A_{\text{base}}(i, j_2)}.
\end{equation}
In Phase 2 ($11,975$ tokens), when 5 distinct 5-digit subsystem axioms were placed side-by-side inside the first 95 tokens of the prompt, Richardson Attention boosted the queried \texttt{LAZARUS-9} axiom to $35.02\%$ while the 4 neighboring axioms combined received only $3.70\%$ ($\sim 0.92\%$ per neighbor---a \textbf{38:1 per-axiom selectivity ratio}), reducing neighbor confusion probability from $0.0006\%$ to $0.0004\%$.

\subsection{The Iso-Potential Cubic Plateau Kernel}
Why do monotonic potentials such as linear ramps ($\Phi \propto d$) or concave logarithms ($\Phi \propto \ln(1+d)$) fail under adversarial multi-hop chaining?
\begin{enumerate}
    \item \textbf{The Linear Cannibalization Trap:} If $\Phi(d) = d / L$, then Zone A ($d \approx 0.99 L$) receives a $1.65\times$ larger logit boost than Zone C ($d \approx 0.60 L$). Inside the softmax exponent, Zone A exponentially cannibalizes intermediate hops (in Phase 3 Config 3, linear tilt boosted Zone A to $50.89\%$ but collapsed Zone B attention by $3.0\times$ and Zone C attention by $12.1\times$).
    \item \textbf{The Concave Logarithmic Decoy Trap:} Because $\ln(1+d)/\ln(1+L)$ is concave, decoys located at $d \in [0.10 L, 0.50 L]$ already receive $70\%\text{--}92\%$ of the maximum plateau boost, inadvertently amplifying mid-context decoys alongside the true registries.
\end{enumerate}

To solve both traps simultaneously, we define the \textbf{Cubic Superconducting Plateau Potential} $\Phi_l(i,j)$ parameterized by a layer-specific horizon cutoff fraction $c_l \in (0, 1]$ and convexity exponent $p = 3$:
\begin{equation}
    \Phi_l(i, j) = \left[\min\!\left(1.0, \, \frac{d_{\text{long}}(i, j)}{c_l \cdot \max(k_{\text{len}},\, H_{\text{cal}})}\right)\right]^p, \qquad p = 3.
    \label{eq:cubic_plateau}
\end{equation}
Equation~\eqref{eq:cubic_plateau} partitions the causal past into two distinct regimes:
\begin{itemize}
    \item \textbf{The Iso-Potential Superconducting Roof ($d_{\text{long}} \ge c_l L$):} Every deep registry beyond the horizon cutoff $c_l L$ sits at the exact same potential $\Phi_l = 1.000$. Because $\Phi_l(\mathcal{Z}_A) = \Phi_l(\mathcal{Z}_B) = \Phi_l(\mathcal{Z}_C) = 1.0$, inter-zone cannibalization is completely eliminated; the registries compete purely on their Q-K semantic resonance $E_{\text{sem}}(i, j;\, \omega)$.
    \item \textbf{The Convex Cubic Attenuation Barrier ($d_{\text{long}} < c_l L$):} For late-context decoys at fraction $u = d_{\text{long}} / (c_l L) < 1$, the potential drops cubically as $u^3$ (e.g., at $u = 0.25$, $\Phi_l = 0.0156$ instead of $0.84$ under logarithmic scaling), starving the decoy region while $-\eta S_{\text{heavy}}$ actively subtracts residual decoy peaks.
\end{itemize}

\subsection{Multi-Hop Composition as a Discrete Renormalization Trajectory}
In an $M$-hop query such as \emph{``What is the Genesis Code for the primary vault commanded by \texttt{DIRECTOR-VANCE}?''}, the initial query embedding $x_i^{(0)}$ contains only the lexical representation of $\text{\texttt{DIRECTOR-VANCE}}$ ($\mathcal{Z}_C$). It does not yet contain the intermediate symbols $\text{\texttt{SECTOR-KRONOS}}$ ($\mathcal{Z}_B$) or $\text{\texttt{VAULT-LAZARUS}}$ ($\mathcal{Z}_A$).

Let $\mathcal{G} = \{l_1, l_2, \dots, l_K\}$ denote the ordered indices of the Global Attention layers (in \texttt{gemma-4-E4B-it}, $\mathcal{G} = \{5, 11, 17, 23, 29, 35, 41\}$). Multi-hop resolution proceeds as a coarse-graining Renormalization Group (RG) flow across $\mathcal{G}$, where the horizon cutoff fraction $c_l$ recedes monotonically with layer depth ($c_{l_1} \le c_{l_2} \le \dots \le c_{l_K}$):
\begin{equation}
    c_l = \begin{cases}
        c_{\text{early}} = 0.52, & l \in \{5, 11, 17\} \quad \text{(Stage 1: Includes }\mathcal{Z}_C, \mathcal{Z}_B, \mathcal{Z}_A \text{ on }\Phi=1 \text{ roof)}, \\
        c_{\text{mid}}   = 0.68, & l \in \{23, 29\}    \quad \text{(Stage 2: Integrates out }\mathcal{Z}_C\text{; holds }\mathcal{Z}_B, \mathcal{Z}_A \text{ on roof)}, \\
        c_{\text{late}}  = 0.85, & l \in \{35, 41\}    \quad \text{(Stage 3: Integrates out }\mathcal{Z}_B\text{; isolates }\mathcal{Z}_A \text{ for readout)}.
    \end{cases}
    \label{eq:rg_schedule}
\end{equation}

\begin{algorithm}[htbp]
\caption{Richardson Attention: Runtime Inference \& Depth-Progressive RG Relay}
\label{alg:richardson}
\begin{algorithmic}[1]
\Require Pre-trained Transformer $\mathcal{M}$ with Global Layers $\mathcal{G}$; Calibration horizon $H_{\text{cal}}$; Quorum size $Q$ per global layer; Query window $W_q = 64$; Gains $(\beta, \eta)$; RG schedule $\{c_l\}_{l \in \mathcal{G}}$.
\Ensure Zero-shot extrapolated forward pass with intact local grammar.
\State \textbf{Stage 1 (Offline 1-Pass Diagnostic Scan):}
\State Run 1 baseline forward pass on calibration text of length $H_{\text{cal}}$.
\For{each head $k = (l, h)$}
    \State Compute $\hat{\alpha}_{l,h} = (\bar{d}_{l,h} + 10^{-6})^{-1}$ and entropy trace $H_k(i) = -\sum_j A^{(k)}_{i,j} \ln A^{(k)}_{i,j}$.
\EndFor
\State Compute $\mathbf{R} = \operatorname{corrcoef}(\mathbf{H})$, Precision Matrix $\mathbf{P} = \mathbf{R}^+$, and Agency $P_{k,k} = \operatorname{diag}(\mathbf{P})$.
\For{each global layer $l \in \mathcal{G}$}
    \State Select top $Q$ heads in layer $l$ ranked by minimal $\hat{\alpha}_{l,h}$ (tie-broken by maximal $P_{k,k}$).
    \State Store calibrated coupling $\kappa_{l,h} = \operatorname{clip}(\hat{\alpha}_{l,h} H_{\text{cal}}, 1.8, 2.6)$ in target map $\mathcal{T}$.
\EndFor
\State \textbf{Stage 2 (Runtime Forward Pass Hook on Layer $l$, Query Length $q_{\text{len}}$, Key Length $k_{\text{len}}$):}
\If{$l \notin \mathcal{G}$}
    \State \Return Standard Sliding-Window Attention (preserves 100\% of local syntactic heads).
\EndIf
\State Compute base scaled dot-product logits $L^{(l,h)} = \frac{Q K^\top}{\sqrt{d_k}} + M_{\text{causal}}$.
\State Identify active query readout slice $i \in [i_{\text{start}}, q_{\text{len}})$ where $i_{\text{start}} = \max(0, q_{\text{len}} - W_q)$.
\State For prefix rows $i < i_{\text{start}}$, compute unmodified $A^{(l,h)}_{i,\cdot} = \operatorname{softmax}(L^{(l,h)}_{i,\cdot})$ (protects KV-Write binding).
\State For active query rows $i \in [i_{\text{start}}, q_{\text{len}})$, compute $d_{\text{long}}(i,j)$ via Eq.~\eqref{eq:d_long} and cubic plateau $\Phi_l(i,j)$ via Eq.~\eqref{eq:cubic_plateau}.
\For{each head $h \in \{0, \dots, H-1\}$ on active rows $i \ge i_{\text{start}}$}
    \If{$(l, h) \in \mathcal{T}$}
        \State $\Delta L_{i,j} \gets \beta\, \kappa_{l,h}\, \Phi_l(i,j)$
        \State $S_{\text{free}} \gets \operatorname{softmax}(L^{(l,h)}_{i,\cdot} + \Delta L_{i,\cdot})$; \quad $S_{\text{heavy}} \gets \operatorname{softmax}(L^{(l,h)}_{i,\cdot} - \Delta L_{i,\cdot})$
        \State $A^{(l,h)}_{i,\cdot} \gets \mathcal{N}\!\big(\max(0, (1+\eta) S_{\text{free}} - \eta S_{\text{heavy}})\big)$
    \Else
        \State $A^{(l,h)}_{i,\cdot} \gets \operatorname{softmax}(L^{(l,h)}_{i,\cdot})$
    \EndIf
\EndFor
\State \Return $A^{(l,h)} V$
\end{algorithmic}
\end{algorithm}

\begin{enumerate}
    \item \textbf{Independent Claim 1 (Diagnostic Scanner via Precision-Matrix VIF \& Spectral Mass):} Automated zero-shot identification of long-range causal driver heads in a pre-trained neural network by computing per-head empirical decay mass $\hat{\alpha}_{l,h} = (\bar{d}_{l,h} + \epsilon)^{-1}$ and the diagonal Variance Inflation Factors $\operatorname{diag}(\mathbf{R}^+)$ of the inter-head entropy correlation matrix.
    \item \textbf{Independent Claim 2 (Write/Read Asymmetric Query-Slice Extrapolation):} A runtime attention inference engine that preserves unmodified local gravity on document prefill rows ($i < T - W_q$) to protect intra-clause KV-cache entity binding while selectively activating spectral extrapolation on the active query/generation slice ($i \in [T - W_q, T)$), subject to token-0 sink pinning ($d_{i,0} = 0$) and a local syntactic guardband $d_{\text{long}}(i,j) = \max(0, i - j - W_{\text{guard}})$.
    \item \textbf{Independent Claim 3 (Dual-State Symmetric Extrapolation with Iso-Potential Cubic Plateau):} Evaluating the attention distribution at symmetric positive and negative potential shifts $S_{\text{free}} = \operatorname{softmax}(L + \Delta L)$ and $S_{\text{heavy}} = \operatorname{softmax}(L - \Delta L)$ over a convex plateau potential $\Phi_l(i,j) = \min(1.0, d_{\text{long}}(i,j) / (c_l L))^p$ ($p \ge 2$), and combining them via $\mathcal{N}(\max(0, (1+\eta)S_{\text{free}} - \eta S_{\text{heavy}}))$ to simultaneously eliminate inter-zone cannibalization on the plateau ($d \ge c_l L$) and cancel mid/late-context decoys ($d < c_l L$).
    \item \textbf{Dependent Claim 4 (Depth-Progressive Renormalization Group Relay):} Stratifying the selected driver heads across multiple Global Attention layers $\mathcal{G} = \{l_1, \dots, l_K\}$ with monotonically increasing horizon cutoffs $c_{l_1} \le c_{l_2} \le \dots \le c_{l_K}$ to execute multi-hop relational composition across network depth in a single forward pass.
\end{enumerate}
